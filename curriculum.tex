% Bullet inside \entry / \project
\newcommand{\point}[1]{\item\small{#1\vspace{-2pt}}}

% \entry{Org}{Role}{Location}{Dates}{bullets}
\newcommand{\entry}[5]{%
  \vspace{-2pt}\item
  \begin{tabular*}{\textwidth}[t]{l@{\extracolsep{\fill}}r}
    \textbf{#1} & \textbf{\small #4} \\
    \textit{\small #2} & \textit{\small #3} \\
  \end{tabular*}\vspace{-7pt}
  \begin{itemize}#5\end{itemize}\vspace{-9pt}
}

% \project{Title}{Org}{Stack}{Dates}{bullets}  (Org may be empty: {})
\newcommand{\project}[5]{%
  \vspace{-2pt}\item
  \begin{tabular*}{1.001\textwidth}{l@{\extracolsep{\fill}}r}
    \small\textbf{#1}\ifx\relax#2\relax\else\ with \textbf{#2}\fi\ \emph{| #3} & \textbf{\small #4}\\
  \end{tabular*}\vspace{-7pt}
  \begin{itemize}#5\end{itemize}\vspace{-9pt}
}

% \skill{Label}{Value}
\newcommand{\skill}[2]{\item \textbf{#1:} #2}

% Wraps a list of \entry or \project items
\newenvironment{entries}
  {\begin{itemize}[leftmargin=0in, label={}]}
  {\end{itemize}}

% Wraps a list of \skill items (compact, indented)
\newenvironment{keyvalues}
  {\begin{itemize}[leftmargin=0.15in, label={}, itemsep=-2pt]}
  {\end{itemize}}
