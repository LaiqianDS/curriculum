% Bullets inside \entry / \project
\setlist[itemize,2]{nosep, topsep=2pt, itemsep=1pt, leftmargin=1.2em, before=\small}
\newcommand{\point}[1]{\item #1}

% \entry{Org}{Role}{Location}{Dates}{bullets}
\newcommand{\entry}[5]{%
  \item \textbf{#1}\hfill{\small #4}\\
  \textit{\small #2}\hfill\textit{\small #3}
  \begin{itemize}#5\end{itemize}
}

% \project{Title}{Org}{Stack}{Dates}{bullets}  (Org may be empty: {})
\newcommand{\project}[5]{%
  \item \textbf{#1}\ifx\relax#2\relax\else\ with \textbf{#2}\fi\ \emph{\small | #3}\hfill{\small #4}
  \begin{itemize}#5\end{itemize}
}

% \skill{Label}{Value}
\newcommand{\skill}[2]{\item \textbf{#1:} #2}

% Wraps a list of \entry or \project items
\newenvironment{entries}
  {\begin{itemize}[leftmargin=0in, label={}, topsep=0pt, partopsep=0pt, parsep=0pt, itemsep=8pt]}
  {\end{itemize}}

% Wraps a list of \skill items (compact)
\newenvironment{keyvalues}
  {\begin{itemize}[leftmargin=0in, label={}, topsep=0pt, partopsep=0pt, parsep=0pt, itemsep=3pt, before=\small]}
  {\end{itemize}}
